1	import { StyleSheet, View, TouchableOpacity, Modal, Pressable, ScrollView } from 'react-native';
2	import { MenuProvider } from 'react-native-popup-menu';
3	import { Feather } from '@expo/vector-icons';
4	import { router } from 'expo-router';
5	import FancyImage from '../images/FancyImage';
6	import FancyText from '../FancyText';
7	import DefaultIcons from '../FancyIcons';
8	import FancyPopup from '../popup/FancyPopup';
9	import { ResponseLoginIgrejaDto } from '../../domain/dtos/login/login.response';
10	import { Pallete } from '../../constants/colors';
11	import { useSairDaIgreja } from '../../hooks/useSairDaIgreja';
12	import { FancyAlert } from '../modal/FancyAlert';
13	
14	interface FancyDrawerIgrejaSelectorModalProps {
15	  visible: boolean;
16	  onClose: () => void;
17	  igrejas: ResponseLoginIgrejaDto[];
18	  igrejaAtiva: ResponseLoginIgrejaDto | null | undefined;
19	  onSelectIgreja: (igreja: ResponseLoginIgrejaDto) => void;
20	}
21	
22	export default function FancyDrawerIgrejaSelectorModal({
23	  visible,
24	  onClose,
25	  igrejas,
26	  igrejaAtiva,
27	  onSelectIgreja,
28	}: FancyDrawerIgrejaSelectorModalProps) {
29	  const { sairDaIgreja, validateRole, isPending } = useSairDaIgreja();
30	
31	  const handleAdicionarIgreja = () => {
32	    onClose();
33	    router.push('/(app)/join-church');
34	  };
35	
36	  const handleMinhasSolicitacoes = () => {
37	    onClose();
38	    router.push('/(app)/join-church/requests');
39	  };
40	
41	  const handleSairDaIgreja = (igreja: ResponseLoginIgrejaDto) => {
42	    if (!validateRole(igreja.role)) return;
43	    FancyAlert.alert(
44	      'Sair da igreja',
45	      `Tem certeza que deseja sair de "${igreja.nome}"?`,
46	      [
47	        { text: 'Cancelar', style: 'cancel' },
48	        {
49	          text: 'Sim',
50	          style: 'destructive',
51	          onPress: () => sairDaIgreja({ igrejaId: igreja.id, role: igreja.role }),
52	        },
53	      ],
54	    );
55	  };
56	
57	  return (
58	    <Modal visible={visible} transparent={true} animationType='fade' onRequestClose={onClose}>
59	      <MenuProvider skipInstanceCheck>
60	        {/* Backdrop */}
61	        <Pressable style={styles.backdrop} onPress={onClose}>
62	          {/* Modal Content */}
63	          <Pressable style={styles.modalContent} onPress={(e) => e.stopPropagation()}>
64	            {/* Header */}
65	            <View style={styles.modalHeader}>
66	              <FancyText size='extraSmall' type='semiBold' color={Pallete.fonts.inactive}>
67	                SUAS IGREJAS
68	              </FancyText>
69	            </View>
70	
71	            {/* Lista de Igrejas */}
72	            <ScrollView style={styles.modalScrollView}>
73	              {igrejas.length === 0 ? (
74	                <View style={styles.emptyContainer}>
75	                  <FancyText type='normal' size='medium' style={styles.emptyText}>
76	                    Nenhuma igreja cadastrada
77	                  </FancyText>
78	                </View>
79	              ) : (
80	                <View style={{ marginVertical: 5 }}>
81	                  {igrejas.map((igreja, index) => {
82	                    const isAtiva = igreja.id === igrejaAtiva?.id;
83	                    const isLast = index === igrejas.length - 1;
84	
85	                    return (
86	                      <View key={igreja.id} style={[styles.igrejaItem, !isLast && styles.igrejaItemBorder]}>
87	                        <TouchableOpacity
88	                          onPress={() => onSelectIgreja(igreja)}
89	                          activeOpacity={0.7}
90	                          style={styles.igrejaItemMain}
91	                        >
92	                          {/* Logo */}
93	                          {igreja.logoThumbUrl || igreja.logoUrl ? (
94	                            <FancyImage
95	                              source={{
96	                                uri: (igreja.logoThumbUrl || igreja.logoUrl) as string,
97	                              }}
98	                              size={28}
99	                              style={styles.igrejaLogo}
100	                            />
101	                          ) : (
102	                            <View style={styles.igrejaLogoPlaceholder}>
103	                              <DefaultIcons.Custom
104	                                library='MaterialCommunityIcons'
105	                                name='church'
106	                                size={19}
107	                                color={Pallete.icons.inactive}
108	                                style={{ marginBottom: 2 }}
109	                              />
110	                            </View>
111	                          )}
112	
113	                          {/* Nome */}
114	                          <FancyText
115	                            size={'small'}
116	                            type='semiBold'
117	                            style={styles.igrejaNomeText}
118	                            numberOfLines={1}
119	                          >
120	                            {igreja.nome}
121	                          </FancyText>
122	                        </TouchableOpacity>
123	
124	                        <View style={styles.igrejaItemActions}>
125	                          {isAtiva && (
126	                            <DefaultIcons.Custom
127	                              library='MaterialCommunityIcons'
128	                              name='check-bold'
129	                              size={16}
130	                              color={Pallete.icons.inactive}
131	                            />
132	                          )}
133	                        <FancyPopup
134	                          disabled={isPending}
135	                          items={[
136	                            {
137	                              label: 'Sair da igreja',
138	                              onPress: () => handleSairDaIgreja(igreja),
139	                              icon: { library: 'Feather', name: 'log-out', size: 16, color: Pallete.fonts.dark },
140	                            },
141	                          ]}
142	                          triggerComponent={
143	                            <View style={styles.menuTrigger}>
144	                              <Feather name='more-vertical' size={18} color={Pallete.icons.inactive} />
145	                            </View>
146	                          }
147	                        />
148	                        </View>
149	                      </View>
150	                    );
151	                  })}
152	                </View>
153	              )}
154	
155	              {/* Divisor */}
156	              <View style={styles.divisor} />
157	
158	              {/* Entrar em uma nova igreja */}
159	              <TouchableOpacity
160	                onPress={handleAdicionarIgreja}
161	                activeOpacity={0.7}
162	                style={styles.actionItem}
163	              >
164	                <DefaultIcons.Custom
165	                  library='MaterialIcons'
166	                  name='login'
167	                  size={16}
168	                  color={Pallete.icons.inactive}
169	                />
170	                <FancyText size='small' type='medium'>
171	                  Entrar em uma nova igreja
172	                </FancyText>
173	              </TouchableOpacity>
174	
175	              {/* Minhas Solicitações */}
176	              <TouchableOpacity
177	                onPress={handleMinhasSolicitacoes}
178	                activeOpacity={0.7}
179	                style={styles.actionItemLast}
180	              >
181	                <DefaultIcons.Custom
182	                  library='MaterialIcons'
183	                  name='history'
184	                  size={16}
185	                  color={Pallete.icons.inactive}
186	                />
187	                <FancyText size='small' type='medium'>
188	                  Minhas solicitações
189	                </FancyText>
190	              </TouchableOpacity>
191	            </ScrollView>
192	          </Pressable>
193	        </Pressable>
194	      </MenuProvider>
195	    </Modal>
196	  );
197	}
198

200	  // MODAL
201	  backdrop: {
202	    flex: 1,
203	    backgroundColor: 'rgba(0, 0, 0, 0.5)',
204	    justifyContent: 'center',
205	    alignItems: 'center',
206	    padding: 16,
207	  },
208	  modalContent: {
209	    width: '100%',
210	    maxWidth: 320,
211	    backgroundColor: '#FFFFFF',
212	    borderRadius: 16,
213	    shadowColor: '#000',
214	    shadowOffset: { width: 0, height: 8 },
215	    shadowOpacity: 0.15,
216	    shadowRadius: 24,
217	    elevation: 8,
218	  },
219	  modalHeader: {
220	    paddingHorizontal: 16,
221	    paddingTop: 16,
222	    paddingBottom: 12,
223	    borderBottomWidth: 1,
224	    borderBottomColor: '#F3F4F6',
225	  },
226	  modalScrollView: {
227	    maxHeight: 400,
228	  },
229	
230	  // EMPTY STATE
231	  emptyContainer: {
232	    paddingVertical: 32,
233	    alignItems: 'center',
234	  },
235	  emptyText: {
236	    fontWeight: '400',
237	    color: '#9CA3AF',
238	  },
239	
240	  // ITEM DA LISTA
241	  igrejaItem: {
242	    height: 52,
243	    paddingHorizontal: 16,
244	    paddingVertical: 12,
245	    flexDirection: 'row',
246	    alignItems: 'center',
247	    backgroundColor: 'transparent',
248	  },
249	  igrejaItemMain: {
250	    flex: 1,
251	    flexDirection: 'row',
252	    alignItems: 'center',
253	  },
254	  igrejaItemActions: {
255	    flexDirection: 'row',
256	    alignItems: 'center',
257	    gap: 8,
258	  },
259	  igrejaItemBorder: {
260	    borderBottomWidth: 1,
261	    borderBottomColor: '#F3F4F6',
262	  },
263	  igrejaLogo: {
264	    ...Pallete.shadows[200],
265	    width: 30,
266	    height: 30,
267	    borderRadius: 999,
268	    borderWidth: 0.2,
269	    borderColor: Pallete.border,
270	    marginRight: 12,
271	  },
272	  igrejaLogoPlaceholder: {
273	    width: 32,
274	    height: 32,
275	    borderRadius: 999,
276	    backgroundColor: '#F3F4F6',
277	    justifyContent: 'center',
278	    alignItems: 'center',
279	    marginRight: 12,
280	  },
281	  igrejaNomeText: {
282	    flex: 1,
283	    opacity: 0.8,
284	  },

293	  },

298	    paddingHorizontal: 16,
299	    paddingVertical: 12,
300	    flexDirection: 'row',
301	    alignItems: 'center',
302	    gap: 10,
303	  },

319	    gap: 10,
320	  },

326	  },
327	});
