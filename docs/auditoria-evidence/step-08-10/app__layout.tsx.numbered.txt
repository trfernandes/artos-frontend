62	function RootLayoutNav() {
63	  const { user, loading } = useAuth();
64	  const segments = useSegments();
65	  const isAuthRoute = segments[0] === '(auth)';
66	  const statusBarStyle: 'light' | 'dark' = isAuthRoute ? 'light' : 'dark';
67	  const safeAreaBackgroundColor = isAuthRoute ? '#FFFFFF' : Pallete.backgroundColor;
68	
69	  const [fontsLoaded] = useFonts({
70	    MontserratBlack: require('../assets/fonts/montserrat/Montserrat-Black.ttf'),
71	    MontserratBlackItalic: require('../assets/fonts/montserrat/Montserrat-BlackItalic.ttf'),
72	    MontserratBold: require('../assets/fonts/montserrat/Montserrat-Bold.ttf'),
73	    MontserratBoldItalic: require('../assets/fonts/montserrat/Montserrat-BoldItalic.ttf'),
74	    MontserratExtraBold: require('../assets/fonts/montserrat/Montserrat-ExtraBold.ttf'),
75	    MontserratExtraBoldItalic: require('../assets/fonts/montserrat/Montserrat-ExtraBoldItalic.ttf'),
76	    MontserratExtraLight: require('../assets/fonts/montserrat/Montserrat-ExtraLight.ttf'),
77	    MontserratExtraLightItalic: require('../assets/fonts/montserrat/Montserrat-ExtraLightItalic.ttf'),
78	    MontserratItalic: require('../assets/fonts/montserrat/Montserrat-Italic.ttf'),
79	    MontserratLight: require('../assets/fonts/montserrat/Montserrat-Light.ttf'),
80	    MontserratLightItalic: require('../assets/fonts/montserrat/Montserrat-LightItalic.ttf'),
81	    MontserratMedium: require('../assets/fonts/montserrat/Montserrat-Medium.ttf'),
82	    MontserratMediumItalic: require('../assets/fonts/montserrat/Montserrat-MediumItalic.ttf'),
83	    MontserratRegular: require('../assets/fonts/montserrat/Montserrat-Regular.ttf'),
84	    MontserratSemiBold: require('../assets/fonts/montserrat/Montserrat-SemiBold.ttf'),
85	    MontserratSemiBoldItalic: require('../assets/fonts/montserrat/Montserrat-SemiBoldItalic.ttf'),
86	    MontserratThin: require('../assets/fonts/montserrat/Montserrat-Thin.ttf'),
87	    MontserratThinItalic: require('../assets/fonts/montserrat/Montserrat-ThinItalic.ttf'),
88	  });

90	  // esconder splash quando tudo estiver pronto
91	  useEffect(() => {

94	    }
95	  }, [fontsLoaded, loading]);

97	  // registrar notificações quando logar

99	    if (user?.user?.id) {
100	      registerForPushNotificationsAsync(user.user.id);
101	    }
102	  }, [user?.user?.id]);
103	
104	  if (!fontsLoaded || loading) {
105	    return null;
106	  }
107	
108	  return (
109	    <FancyAlertProvider>
110	      <View style={styles.rootContainer}>
111	        <NotificationsManager />
112	        {user && <AppReviewManager />}
113	        <StatusBar
114	          backgroundColor='transparent'
115	          style={statusBarStyle}

117	        />
118	        <SafeAreaView
119	          style={[styles.navigationContainer, { backgroundColor: safeAreaBackgroundColor }]}

123	            <Stack.Screen name='(app)' options={{ headerShown: false }} />

126	          </Stack>
127	        </SafeAreaView>
128	        <Toast config={toastConfig} position='bottom' visibilityTime={4000} />
129	        <FancyAlertConnector />
130	        <ConnectivityBanner />
131	      </View>
132	    </FancyAlertProvider>
133	  );
