1	import axios from 'axios';
2	
3	function parseMessage(message: unknown): string | null {
4	  if (typeof message === 'string' && message.trim()) return message.trim();
5	  if (Array.isArray(message)) {
6	    const joined = message.filter((item) => typeof item === 'string' && item.trim()).join(', ');
7	    return joined || null;
8	  }
9	  return null;
10	}
11	
12	export function getApiErrorMessage(error: unknown, fallback = 'Erro inesperado'): string {
13	  if (axios.isAxiosError(error)) {
14	    const status = error.response?.status;
15	    const data: any = error.response?.data;
16	
17	    const message =
18	      parseMessage(data?.message) ||
19	      parseMessage(data?.error?.message) ||
20	      parseMessage(error.message);
21	
22	    if (message) return message;
23	
24	    if (status === 401) return 'Sessão expirada. Faça login novamente.';
25	    if (status === 403) return 'Você não tem permissão para acessar este recurso.';
26	    if (status === 400) return 'Requisição inválida. Verifique os dados informados.';
27	  }
28	
29	  if (error instanceof Error && error.message.trim()) {
30	    return error.message.trim();
31	  }
32	
33	  return fallback;
34	}
