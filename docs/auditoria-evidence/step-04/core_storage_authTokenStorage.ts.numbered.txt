1	import AsyncStorage from '@react-native-async-storage/async-storage';
2	import * as SecureStore from 'expo-secure-store';
3	
4	const TOKEN_KEY = 'token';
5	
6	export async function getAuthToken(): Promise<string | null> {
7	  const secureToken = await SecureStore.getItemAsync(TOKEN_KEY);
8	  if (secureToken) return secureToken;
9	
10	  const legacyToken = await AsyncStorage.getItem(TOKEN_KEY);
11	  if (legacyToken) {
12	    await SecureStore.setItemAsync(TOKEN_KEY, legacyToken);
13	    await AsyncStorage.removeItem(TOKEN_KEY);
14	  }
15	
16	  return legacyToken;
17	}
18	
19	export async function setAuthToken(token: string): Promise<void> {
20	  await SecureStore.setItemAsync(TOKEN_KEY, token);
21	  await AsyncStorage.removeItem(TOKEN_KEY);
22	}
23	
24	export async function clearAuthToken(): Promise<void> {
25	  await SecureStore.deleteItemAsync(TOKEN_KEY);
26	  await AsyncStorage.removeItem(TOKEN_KEY);
27	}
