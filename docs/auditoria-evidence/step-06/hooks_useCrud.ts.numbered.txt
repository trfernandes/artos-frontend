1	import { useQueryClient, useQuery, useMutation } from '@tanstack/react-query';
2	import { useState, useRef, useMemo, useEffect } from 'react';
3	import { FieldValues, Resolver, UseFormReturn, useForm } from 'react-hook-form';
4	import Toast from 'react-native-toast-message';
5	import { DynamicQuery } from '../domain/utils/query_utils';
6	
7	export type CrudFormMessages = {
8	  successCreate?: string;
9	  successUpdate?: string;
10	  successDelete?: string;
11	  errorCreate?: string;
12	  errorUpdate?: string;
13	  errorDelete?: string;
14	};
15	
16	type CrudQueryKind = 'none' | 'query' | 'id';
17	
18	export type CrudIdentifiable = { id?: string | null };
19	
20	export interface ExternalUseCrudParams {
21	  autoFetch?: boolean;
22	  initialParams?: DynamicQuery | string | null;
23	  messages?: CrudFormMessages;
24	  muteMessages?: boolean;
25	}
26	
27	export interface UseCrudOptions<TResponse extends CrudIdentifiable, TForm extends FieldValues, TCreate, TUpdate> {
28	  queryKey: string;
29	
30	  fetchAll: () => Promise<TResponse[]>;
31	  fetchOne?: (id: string) => Promise<TResponse>;
32	
33	  add?: (data: TCreate) => Promise<TResponse>;
34	  update?: (id: string, data: TUpdate) => Promise<TResponse>;
35	  remove?: (id: string) => Promise<void>;
36	
37	  search?: (params: DynamicQuery) => Promise<TResponse[]>;
38	
39	  resolver?: Resolver<TForm, any, TForm>;
40	  messages?: CrudFormMessages;
41	  muteMessages?: boolean;
42	
43	  /**
44	   * Se o TForm não for exatamente o CreateDto/UpdateDto, use estes mappers.
45	   * Por padrão, faz cast direto.
46	   */
47	  toCreateDto?: (form: TForm) => TCreate;
48	  toUpdateDto?: (form: TForm, current: TResponse) => TUpdate;
49	}
50	
51	export function useCrud<TResponse extends CrudIdentifiable, TForm extends FieldValues, TCreate = TForm, TUpdate = Partial<TForm>>({
52	  queryKey,
53	  fetchAll,
54	  fetchOne,
55	  add,
56	  update,
57	  remove,
58	  search,
59	  resolver,
60	  muteMessages = false,
61	  messages,
62	  autoFetch = false,
63	  initialParams,
64	  toCreateDto,
65	  toUpdateDto,
66	}: UseCrudOptions<TResponse, TForm, TCreate, TUpdate> & ExternalUseCrudParams) {
67	  const queryClient = useQueryClient();
68	
69	  const [currentItem, setCurrentItem] = useState<TResponse | null>(null);
70	
71	  const [searchParams, setSearchParams] = useState<DynamicQuery | null>(typeof initialParams === 'string' ? null : initialParams ?? null);
72	  const [hasBootstrappedInitialParams, setHasBootstrappedInitialParams] = useState(() => !initialParams);
73	  const [isFetchingInitialItem, setIsFetchingInitialItem] = useState(false);
74	
75	  const appliedInitialParamsKeyRef = useRef<string | null>(null);
76	
77	  const initialParamsSnapshot = useMemo((): {
78	    key: string | null;
79	    kind: CrudQueryKind;
80	    value?: DynamicQuery | string;
81	  } => {
82	    if (!initialParams) return { key: null, kind: 'none' };
83	
84	    if (typeof initialParams === 'string') {
85	      return { key: `id:${initialParams}`, kind: 'id', value: initialParams };
86	    }
87	
88	    let serialized = '';
89	    try {
90	      serialized = JSON.stringify(initialParams);
91	    } catch {
92	      serialized = JSON.stringify({ ...initialParams });
93	    }
94	
95	    return { key: `query:${serialized}`, kind: 'query', value: initialParams };
96	  }, [initialParams]);
97	
98	  const form: UseFormReturn<TForm> = useForm<TForm>({ resolver });
99	
100	  useEffect(() => {
101	    const { key, kind, value } = initialParamsSnapshot;
102	
103	    if (!key) {
104	      if (appliedInitialParamsKeyRef.current !== key) {
105	        appliedInitialParamsKeyRef.current = key;
106	        setSearchParams((prev) => (prev === null ? prev : null));
107	      }
108	      setHasBootstrappedInitialParams(true);
109	      return;
110	    }
111	
112	    if (appliedInitialParamsKeyRef.current === key) return;
113	    appliedInitialParamsKeyRef.current = key;
114	
115	    if (kind === 'id') {
116	      if (!fetchOne || typeof value !== 'string') {
117	        setHasBootstrappedInitialParams(true);
118	        return;
119	      }
120	
121	      let isMounted = true;
122	      setIsFetchingInitialItem(true);
123	
124	      fetchOne(value)
125	        .then((item) => {
126	          if (!isMounted || !item) return;
127	          setCurrentItem(item);
128	        })
129	        .finally(() => {
130	          if (!isMounted) return;
131	          setIsFetchingInitialItem(false);
132	          setHasBootstrappedInitialParams(true);
133	        });
134	
135	      return () => {
136	        isMounted = false;
137	      };
138	    }
139	
140	    if (kind === 'query' && value && typeof value !== 'string') {
141	      setSearchParams(value);
142	    }
143	
144	    setHasBootstrappedInitialParams(true);
145	  }, [initialParamsSnapshot, fetchOne]);
146	
147	  const dataQuery = useQuery<TResponse[], Error>({
148	    queryKey: [queryKey, searchParams],
149	    queryFn: async () => {
150	      if (searchParams && search) return await search(searchParams);
151	      return await fetchAll();
152	    },
153	    enabled: hasBootstrappedInitialParams && (autoFetch || searchParams !== null),
154	  });
155	
156	  const hasReceivedData = dataQuery.data !== undefined;
157	  const combinedLoading =
158	    !hasBootstrappedInitialParams || (!hasReceivedData && (dataQuery.isLoading || dataQuery.isFetching)) || isFetchingInitialItem;
159	  const combinedRefetching = hasReceivedData && dataQuery.isFetching && !dataQuery.isLoading;
160	
161	  const createMutation = useMutation({
162	    mutationFn: add,
163	    onSuccess: () => {
164	      if (!muteMessages) Toast.show({ type: 'success', text1: messages?.successCreate || 'Item criado com sucesso!' });
165	      queryClient.invalidateQueries({ queryKey: [queryKey] });
166	    },
167	    onError: (error) => {
168	      if (!muteMessages) Toast.show({ type: 'error', text1: messages?.errorCreate || 'Erro ao criar item.' });
169	      if (__DEV__) {
170	        console.log('[useCrud] Create error:', error);
171	      }
172	    },
173	  });
174	
175	  const updateMutation = update && useMutation({
176	    mutationFn: ({ id, data }: { id: string; data: TUpdate }) => update(id, data),
177	    onSuccess: () => {
178	      if (!muteMessages) Toast.show({ type: 'success', text1: messages?.successUpdate || 'Item atualizado com sucesso!' });
179	      queryClient.invalidateQueries({ queryKey: [queryKey] });
180	    },
181	    onError: (error) => {
182	      if (!muteMessages) Toast.show({ type: 'error', text1: messages?.errorUpdate || 'Erro ao atualizar item.' });
183	      if (__DEV__) {
184	        console.log('[useCrud] Update error:', error);
185	      }
186	    },
187	  });
188	
189	  const removeMutation = useMutation({
190	    mutationFn: remove,
191	    onSuccess: () => {
192	      if (!muteMessages) Toast.show({ type: 'success', text1: messages?.successDelete || 'Item removido com sucesso!' });
193	      queryClient.invalidateQueries({ queryKey: [queryKey] });
194	    },
195	    onError: (error) => {
196	      if (!muteMessages) Toast.show({ type: 'error', text1: messages?.errorDelete || 'Erro ao remover item.' });
197	      if (__DEV__) {
198	        console.log('[useCrud] Delete error:', error);
199	      }
200	    },
201	  });
202	
203	  const handleSubmit = async (formValues: TForm) => {
204	    if (currentItem?.id) {
205	      const payload = toUpdateDto ? toUpdateDto(formValues, currentItem) : (formValues as unknown as TUpdate);
206	      await updateMutation?.mutateAsync({ id: currentItem.id, data: payload });
207	    } else {
208	      const payload = toCreateDto ? toCreateDto(formValues) : (formValues as unknown as TCreate);
209	      await createMutation.mutateAsync(payload);
210	    }
211	  };
212	
213	  return {
214	    queryClient,
215	    queryKey,
216	    form,
217	    data: dataQuery.data || [],
218	    currentItem,
219	    setCurrentItem,
220	    setSearchParams,
221	
222	    add: createMutation.mutateAsync,
223	    update: updateMutation?.mutateAsync,
224	    remove: removeMutation.mutateAsync,
225	
226	    handleSubmit,
227	
228	    isLoading: combinedLoading,
229	    isLoadingMutation: createMutation.isPending || updateMutation?.isPending || removeMutation.isPending,
230	    isRefetching: combinedRefetching,
231	    isError: dataQuery.isError,
232	    error: dataQuery.error || createMutation.error || updateMutation?.error || removeMutation.error,
233	    refetch: dataQuery.refetch,
234	  };
235	}
