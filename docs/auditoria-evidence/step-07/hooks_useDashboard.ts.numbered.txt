1	import { useQuery } from '@tanstack/react-query';
2	import { DashboardRepository } from '../domain/services/DashboardRepository';
3	import { useAuth } from '../contexts/AuthContext';
4	import { ResponseDashboardDto } from '../domain/dtos/Dashboard/dashboard.response';
5	import { getApiErrorMessage } from '../domain/api/api-error';
6	
7	const EMPTY_DASHBOARD: ResponseDashboardDto = {
8	  proximasEscalas: [],
9	  totalEscalasMes: 0,
10	  escalasConfirmadas: 0,
11	  escalasPendentes: 0,
12	  proximosEventosMinisterio: [],
13	  solicitacoesPendentes: [],
14	  totalMinisterios: 0,
15	  totalVoluntarios: 0,
16	  totalEventosMes: 0,
17	  ministeriosStats: [],
18	  proximosEventosIgreja: [],
19	  solicitacoesGerais: [],
20	};
21	
22	export function useDashboard() {
23	  const { igrejaAtiva, user } = useAuth();
24	
25	  const query = useQuery({
26	    queryKey: ['dashboard', igrejaAtiva?.id, igrejaAtiva?.role, user?.user?.id],
27	    queryFn: () => {
28	      if (!igrejaAtiva?.id) throw new Error('Igreja não selecionada');
29	      if (!user?.user?.id) throw new Error('Usuário não autenticado');
30	      return DashboardRepository.getDashboard({
31	        igrejaId: igrejaAtiva.id,
32	        userId: user.user.id,
33	        role: igrejaAtiva.role,
34	        ministeriosUsuario: igrejaAtiva.ministerios ?? [],
35	      });
36	    },
37	    enabled: !!igrejaAtiva?.id && !!user?.user?.id,
38	    staleTime: 1000 * 60 * 5, // 5 minutos
39	    retry: 1,
40	  });
41	
42	  const hasServerData = query.data !== undefined;
43	  const errorMessage = query.error ? getApiErrorMessage(query.error, 'Não foi possível carregar o dashboard.') : null;
44	
45	  const data: ResponseDashboardDto = {
46	    ...EMPTY_DASHBOARD,
47	    ...(query.data ?? {}),
48	  };
49	
50	  return {
51	    ...query,
52	    data,
53	    hasServerData,
54	    errorMessage,
55	  };
56	}
57	
58	export function useMinisterioDashboard(ministerioId?: string) {
59	  return useQuery({
60	    queryKey: ['ministerio-dashboard', ministerioId],
61	    queryFn: () => {
62	      if (!ministerioId) throw new Error('Ministério não selecionado');
63	      return DashboardRepository.getMinisterioDashboard(ministerioId);
64	    },
65	    enabled: !!ministerioId,
66	    staleTime: 1000 * 60 * 5,
67	  });
68	}
