1	import { StyleSheet, View } from 'react-native';
2	import FancyButton from '../../components/buttons/FancyButton';
3	import FancyText from '../../components/FancyText';
4	import AuthScreen from '../../components/pages/login/AuthScreen';
5	import { router } from 'expo-router';
6	import { useForm } from 'react-hook-form';
7	import { zodResolver } from '@hookform/resolvers/zod';
8	import ControlledTextInput from '../../components/forms/ControlledTextInput';
9	import ControlledPasswordInput from '../../components/forms/ControlledPasswordInput';
10	import { AxiosError } from 'axios';
11	import Toast from 'react-native-toast-message';
12	import FancyVerticalSpacer from '../../components/FancyVerticalSpacer';
13	import { useVoluntariosCrud } from '../../hooks/useVoluntariosCrud';
14	import { createAccountSchema } from '../../domain/schemas/voluntarioSchema';
15	import { EXTRA_LARGE_SIZE_FONT, LARGE_SIZE_FONT, MEDIUM_SIZE_FONT, SMALL_SIZE_FONT } from '../../constants/font';
16	import { ScrollView } from 'react-native-gesture-handler';
17	import { useConnectivity } from '../../core/network/connectivity/ConnectivityProvider';
18	
19	export default function CreateVoluntarioAccountPage() {
20	  const { status: connectivityStatus } = useConnectivity();
21	  const isServerUnavailable = connectivityStatus !== 'ok';
22	
23	  const { add, isLoadingMutation } = useVoluntariosCrud({
24	    autoFetch: false,
25	    initialParams: null,
26	  });
27	
28	  const createForm = useForm({
29	    resolver: zodResolver(createAccountSchema),
30	    defaultValues: { nome: '', email: '', senha: '', confirmarSenha: '', codigoIgreja: '' },
31	  });
32	
33	  const handleSubmit = createForm.handleSubmit(async (data) => {
34	    try {
35	      const payload: any = {
36	        nome: data.nome,
37	        email: data.email,
38	        senha: data.senha,
39	      };
40	      
41	      // Adicionar código da igreja se fornecido
42	      if (data.codigoIgreja?.trim()) {
43	        payload.codigoIgreja = data.codigoIgreja.trim();
44	      }
45	      
46	      try {
47	        const result = await add(payload);
48	        
49	        // Backend pode retornar informações sobre o join da igreja na resposta
50	        // Se houver erro, mostrar mas não bloquear o fluxo
51	        const anyResult = result as any;
52	        if (anyResult?.igrejaJoinError) {
53	          Toast.show({
54	            type: 'error',
55	            text1: 'Conta criada',
56	            text2: anyResult.igrejaJoinError,
57	            visibilityTime: 5000,
58	          });
59	        } else if (data.codigoIgreja?.trim()) {
60	          Toast.show({
61	            type: 'success',
62	            text1: 'Conta criada com sucesso!',
63	            text2: 'Faça login para acessar.',
64	          });
65	        } else {
66	          Toast.show({
67	            type: 'success',
68	            text1: 'Conta criada com sucesso!',
69	          });
70	        }
71	      } catch (error: AxiosError | any) {
72	        Toast.show({
73	          type: 'error',
74	          text1: 'Não foi possível criar a conta',
75	          text2: error?.response?.data?.message || 'Ocorreu um erro ao criar a conta. Tente novamente mais tarde.',
76	        });
77	        return;
78	      }
79	      
80	      router.replace('/(auth)/login');
81	    } catch (error) {
82	      console.log(error);
83	    }
84	  });
85	
86	  return (
87	    <AuthScreen
88	      showBackButton
89	      disableScroll
90	      scrollContainerStyle={styles.scrollContainer}
91	      headerContainerStyle={styles.titleContainer}
92	      headerWidth={{ default: '85%', keyboard: '110%' }}
93	      contentWidth={{ default: '85%', keyboard: '110%' }}
94	      paddingTopOnKeyboard={60}
95	      alignTopOnKeyboard
96	      fieldsContainerStyle={({ keyboardVisible }) => {
97	        if (keyboardVisible) {
98	          return { height: '85%' };
99	        }
100	      }}
101	      keyboardBottomSpacing={20}
102	      header={({ keyboardVisible }) => (
103	        <View style={{ justifyContent: 'center', gap: 2 }}>
104	          <FancyText
105	            size={!keyboardVisible ? 'extraLarge' : 'large'}
106	            type='bold'
107	            color='white'
108	            style={{ lineHeight: !keyboardVisible ? EXTRA_LARGE_SIZE_FONT * 1.2 : LARGE_SIZE_FONT * 1.2 }}
109	          >
110	            Criação de Conta
111	          </FancyText>
112	          <FancyText
113	            size={!keyboardVisible ? 'medium' : 'small'}
114	            type='medium'
115	            color='white'
116	            style={{
117	              lineHeight: !keyboardVisible ? MEDIUM_SIZE_FONT * 1.3 : SMALL_SIZE_FONT * 1.3,
118	            }}
119	          >
120	            Crie uma conta e aproveite todas as funcionalidades
121	          </FancyText>
122	        </View>
123	      )}
124	    >
125	      {({ keyboardVisible }) => (
126	        <View style={{ gap: 10, flexGrow: 1, flexShrink: 1 }}>
127	          <ScrollView contentContainerStyle={{ gap: 10 }} showsVerticalScrollIndicator={false}>
128	            <ControlledTextInput label='Nome' name='nome' control={createForm.control} />
129	            <ControlledTextInput label='E-mail' name='email' control={createForm.control} inputProps={{ autoCapitalize: 'none' }} />
130	            <ControlledPasswordInput label='Senha' name='senha' control={createForm.control} inputProps={{ secureTextEntry: true }} />
131	            <ControlledPasswordInput
132	              label='Confirmar a Senha'
133	              name='confirmarSenha'
134	              control={createForm.control}
135	              inputProps={{ secureTextEntry: true }}
136	            />
137	            <ControlledTextInput 
138	              label='Código da igreja (opcional)' 
139	              name='codigoIgreja' 
140	              control={createForm.control} 
141	              inputProps={{ 
142	                autoCapitalize: 'none',
143	                placeholder: 'Digite o código se tiver'
144	              }} 
145	            />
146	          </ScrollView>
147	          <FancyVerticalSpacer height={1} />
148	          <FancyButton
149	            label={isLoadingMutation ? 'Confirmando...' : 'Confirmar'}
150	            onPress={handleSubmit}
151	            disabled={isLoadingMutation || isServerUnavailable}
152	            icon={{ library: 'Feather', name: 'check', size: 16 }}
153	          />
154	        </View>
155	      )}
156	    </AuthScreen>
157	  );
158	}
159	
160	const styles = StyleSheet.create({
161	  scrollContainer: {
162	    flexGrow: 1,
163	    paddingHorizontal: 40,
164	    justifyContent: 'center',
165	  },
166	  titleContainer: {
167	    borderColor: 'magenta',
168	    justifyContent: 'center',
169	    // borderWidth: 1,
170	  }, 
171	});
