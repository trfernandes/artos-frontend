1	import { StyleSheet } from 'react-native';
2	import FancyButton from '../../components/buttons/FancyButton';
3	import FancyText from '../../components/FancyText';
4	import AuthScreen from '../../components/pages/login/AuthScreen';
5	import { Pallete } from '../../constants/colors';
6	import Toast from 'react-native-toast-message';
7	import z from 'zod';
8	import ControlledTextInput from '../../components/forms/ControlledTextInput';
9	import { zodResolver } from '@hookform/resolvers/zod';
10	import { useForm } from 'react-hook-form';
11	import { useAuth } from '../../contexts/AuthContext';
12	
13	const forgotPasswordSchema = z.object({
14	  email: z
15	    .string()
16	    .min(1, 'Informe o e-mail')
17	    .regex(/^[^\s@]+@[^\s@]+\.[^\s@]+$/, 'O formato é inválido'),
18	});
19	
20	type ForgotPasswordFormData = z.infer<typeof forgotPasswordSchema>;
21	
22	export default function ForgotPasswordPage() {
23	  const { forgotPassword } = useAuth();
24	
25	  const {
26	    control,
27	    handleSubmit,
28	    formState: { isSubmitting },
29	  } = useForm<ForgotPasswordFormData>({
30	    resolver: zodResolver(forgotPasswordSchema),
31	    defaultValues: { email: '' },
32	  });
33	
34	  const onSubmit = async (data: ForgotPasswordFormData) => {
35	    try {
36	      const success = await forgotPassword(data.email);
37	
38	      if (success) {
39	        Toast.show({
40	          type: 'success',
41	          text1: 'Se o e-mail existir, enviamos instruções de recuperação.',
42	        });
43	      } else {
44	        Toast.show({ type: 'error', text1: 'Erro ao solicitar recuperação.' });
45	      }
46	    } catch {
47	      Toast.show({ type: 'error', text1: 'Erro ao solicitar recuperação.' });
48	    }
49	  };
50	
51	  return (
52	    <AuthScreen
53	      showBackButton
54	      scrollContainerStyle={styles.scrollContainer}
55	      headerContainerStyle={styles.titleContainer}
56	      headerWidth={{ default: '85%', keyboard: '100%' }}
57	      contentWidth={{ default: '85%', keyboard: '100%' }}
58	      paddingTopOnKeyboard={60}
59	      fieldsContainerStyle={styles.fieldsContainer}
60	      header={({ keyboardVisible }) => (
61	        <>
62	          <FancyText size={!keyboardVisible ? 'extraLarge' : 'large'} type='bold' color='white'>
63	            Recuperação de Senha
64	          </FancyText>
65	          <FancyText
66	            size={!keyboardVisible ? 'medium' : 'small'}
67	            type='medium'
68	            color='white'
69	            style={{
70	              width: '99%',
71	              borderWidth: 0,
72	            }}
73	          >
74	            Informe seu e-mail para receber as instruções de recuperação.
75	          </FancyText>
76	        </>
77	      )}
78	    >
79	      <>
80	        <ControlledTextInput
81	          label='E-mail'
82	          name='email'
83	          control={control}
84	          inputProps={{ autoCapitalize: 'none', keyboardType: 'email-address' }}
85	        />
86	
87	        <FancyButton label={isSubmitting ? 'Enviando...' : 'Enviar'} onPress={handleSubmit(onSubmit)} disabled={isSubmitting} />
88	      </>
89	    </AuthScreen>
90	  );
91	}
92	
93	const DESIGN_MODE = 0;
94	
95	const styles = StyleSheet.create({
96	  scrollContainer: {
97	    flexGrow: 1,
98	    paddingHorizontal: 40,
99	    paddingVertical: 0,
100	    justifyContent: 'center',
101	    borderWidth: DESIGN_MODE,
102	    borderColor: 'blueviolet',
103	    gap: 25,
104	  },
105	  titleContainer: {
106	    gap: 2,
107	    borderWidth: DESIGN_MODE,
108	    borderColor: 'magenta',
109	    justifyContent: 'center',
110	  },
111	  fieldsContainer: {
112	    borderWidth: DESIGN_MODE,
113	    borderRadius: 15,
114	    borderColor: 'firebrick',
115	    padding: 25,
116	    gap: 25,
117	    backgroundColor: Pallete.backgroundColor,
118	    ...Pallete.shadows[200],
119	  },
120	});
