1	import { useCallback, useState } from 'react';
2	import { FormProvider, useFieldArray, useForm, useFormContext } from 'react-hook-form';
3	import { DefaultIconsNames } from '../../../../constants/icons';
4	
5	import { EscalaTemplateFormData, EscalaTemplateFuncaoFormData, escalaTemplateFuncaoSchema } from '../../../../domain/schemas/escalaTemplateSchema';
6	import FancyContainerList from '../../../container_list/FancyContainerList';
7	import { zodResolver } from '@hookform/resolvers/zod';
8	import TemplateFuncoesForm from './TemplateFuncoesForm';
9	import { DropDownItemProps } from '../../../fields/FancyDropDownItem';
10	import { Pallete } from '../../../../constants/colors';
11	import { FancyCard } from '../../../cards/Horizontal/FancyCard';
12	import Toast from 'react-native-toast-message';
13	import { FancyAlert } from '../../../modal/FancyAlert';
14	import {
15	    EscalaTemplateExperienciaEnum,
16	    EscalaTemplateExperienciaLabel,
17	} from '../../../../domain/enums/EscalaTemplate/escala-template-experiencia.enum';
18	import { useFuncoesDoMinisterio } from '../../../../hooks/useFuncoesDoMinisterio';
19	import { FancyTextDisplayCard } from '../../../cards/FancyTextDisplayCard';
20	import FancyLoading from '../../../FancyLoading';
21	
22	interface TemplateFuncoesListProps {
23	  disabled?: boolean;
24	  funcoesList?: DropDownItemProps<string>[];
25	  ministerioId: string;
26	}
27	
28	const FORM_DEFAULT_VALUES: Partial<EscalaTemplateFuncaoFormData> = {
29	  experiencia: EscalaTemplateExperienciaEnum.Iniciante,
30	  quantidade: 1,
31	};
32	
33	export default function TemplateFuncoesList({
34	  disabled = false,
35	  funcoesList = [] as DropDownItemProps<string>[],
36	  ministerioId,
37	}: TemplateFuncoesListProps) {
38	  const { control, watch, setValue } = useFormContext<EscalaTemplateFormData>();
39	  const { append: addFuncao, update: updateFuncao, remove: removeFuncao } = useFieldArray({ control, name: 'funcoes' });
40	  const funcoesWatch = watch('funcoes') ?? [];
41	
42	  const [formParams, setFormParams] = useState<{ visible: boolean; mode: 'add' | 'edit' }>();
43	  const formAdd = useForm<EscalaTemplateFuncaoFormData>({
44	    resolver: zodResolver(escalaTemplateFuncaoSchema),
45	    defaultValues: FORM_DEFAULT_VALUES,
46	  });
47	
48	  const { funcoesList: funcoesDataList, isLoading: isLoadingFuncoes } = useFuncoesDoMinisterio(ministerioId);
49	
50	  const handleOpen = useCallback(
51	    (mode: 'add' | 'edit') => {
52	      if (disabled) {
53	        return;
54	      }
55	      formAdd.reset(FORM_DEFAULT_VALUES);
56	      setFormParams({ visible: true, mode });
57	    },
58	    [disabled, formAdd],
59	  );
60	
61	  const handleConfirm = useCallback(
62	    (mode: 'add' | 'edit') => {
63	      if (disabled) {
64	        return;
65	      }
66	      formAdd.handleSubmit((data) => {
67	        if (mode === 'add') {
68	          const alreadyExists = funcoesWatch.some((item) => item.funcaoId === data.funcaoId && item.experiencia === data.experiencia);
69	          if (alreadyExists) {
70	            formAdd.setError('funcaoId', { message: 'Função/Experiência já adicionada.' });
71	            return;
72	          }
73	
74	          addFuncao({
75	            id: data.id,
76	            funcaoId: data.funcaoId,
77	            experiencia: data.experiencia,
78	            quantidade: data.quantidade,
79	          });
80	
81	          Toast.show({
82	            type: 'success',
83	            text1: 'Função adicionada com sucesso!',
84	          });
85	        } else if (mode === 'edit') {
86	          const index = funcoesWatch.findIndex((item) => item.funcaoId === data.funcaoId);
87	          if (index !== -1) {
88	            setValue(`funcoes.${index}`, data);
89	          }
90	          updateFuncao(index, { ...data });
91	
92	          Toast.show({
93	            type: 'success',
94	            text1: 'Função atualizada com sucesso!',
95	          });
96	        }
97	
98	        setFormParams({ visible: false, mode: 'add' });
99	        formAdd.reset(FORM_DEFAULT_VALUES);
100	      })();
101	    },
102	    [addFuncao, disabled, formAdd, funcoesWatch, setValue, updateFuncao],
103	  );
104	
105	  const handleEdit = useCallback(
106	    (index: number) => {
107	      if (disabled) {
108	        return;
109	      }
110	      const entry = funcoesWatch[index];
111	      if (!entry) {
112	        return;
113	      }
114	
115	      formAdd.reset({ ...entry });
116	      setFormParams({ visible: true, mode: 'edit' });
117	    },
118	    [disabled, formAdd, funcoesWatch],
119	  );
120	
121	  const handleRemove = useCallback(
122	    (index: number) => {
123	      if (disabled) {
124	        return;
125	      }
126	      const entry = funcoesWatch[index];
127	      if (!entry) {
128	        return;
129	      }
130	
131	      FancyAlert.alert('Confirmar remoção', `Tem certeza que deseja remover a função?`, [
132	        { text: 'Não', style: 'destructive' },
133	        {
134	          text: 'Sim',
135	          onPress: () => {
136	            removeFuncao(index);
137	            Toast.show({
138	              type: 'success',
139	              text1: 'Função removida com sucesso!',
140	            });
141	          },
142	        },
143	      ]);
144	    },
145	    [disabled, funcoesWatch, removeFuncao],
146	  );
147	
148	  if(isLoadingFuncoes) return <FancyLoading />;
149	
150	  return (
151	    <>
152	      <FancyContainerList
153	        title={'Formação da Equipe'}
154	        data={funcoesWatch}
155	        contentContainerStyle={{ paddingTop: 6 }}
156	        disabled={disabled}
157	        renderItem={({ item, index }: { item: EscalaTemplateFuncaoFormData; index: number }) => {
158	          const matchedOption = funcoesDataList.find((funcao) => funcao.id === item.funcaoId);
159	          const funcaoNome = matchedOption?.nome || item.funcao?.nome || 'Função não encontrada';
160	          const experienciaLabel = EscalaTemplateExperienciaLabel[item.experiencia];
161	          return (
162	            <FancyCard.Simple
163	              title={funcaoNome}
164	              subtitle={<FancyTextDisplayCard icon={{ library: 'MaterialCommunityIcons', name: 'star-outline', size: 12, color: Pallete.primary }} value={experienciaLabel} />}
165	              additionalData1={<FancyTextDisplayCard icon={{ library: 'MaterialCommunityIcons', name: 'account-multiple-outline', size: 12, color: Pallete.primary }} value={item.quantidade.toString()} />}
166	              letter={funcaoNome.charAt(0)}
167	              actionButtons={[
168	                {
169	                  icon: {
170	                    ...DefaultIconsNames.edit,
171	                    backgroundColor: Pallete.primary,
172	                    size: 16,
173	                  },
174	                  onPress: disabled ? undefined : () => handleEdit(index),
175	                },
176	                {
177	                  icon: {
178	                    ...DefaultIconsNames.delete,
179	                    size: 16,
180	                    backgroundColor: Pallete.error,
181	                  },
182	                  onPress: disabled ? undefined : () => handleRemove(index),
183	                },
184	              ]}
185	            />
186	          );
187	        }}
188	        buttons={[
189	          {
190	            icon: { ...DefaultIconsNames.add, size: 20, style: { width: 20, height: 20 } },
191	            onPress: disabled ? undefined : () => handleOpen('add'),
192	          },
193	        ]}
194	      />
195	      <FormProvider {...formAdd}>
196	        {formParams?.visible && (
197	          <TemplateFuncoesForm
198	            mode={formParams.mode}
199	            modalProps={{ visible: formParams.visible }}
200	            onButton1Press={() => {
201	              formAdd.reset(FORM_DEFAULT_VALUES);
202	              setFormParams({ visible: false, mode: 'add' });
203	            }}
204	            onButton2Press={() => handleConfirm(formParams.mode)}
205	            funcoesList={funcoesList}
206	          />
207	        )}
208	      </FormProvider>
209	    </>
210	  );
211	}
