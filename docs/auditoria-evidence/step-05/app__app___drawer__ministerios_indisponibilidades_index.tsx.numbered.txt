1	import { useLocalSearchParams } from 'expo-router';
2	import FancyPageView from '../../../../../components/containers/FancyPageView';
3	import { StyleSheet, View } from 'react-native';
4	import { useForm } from 'react-hook-form';
5	import z from 'zod';
6	import ControlledSearchSelect from '../../../../../components/forms/ControlledSearchSelect';
7	import { useVoluntariosDoMinisterioCrud } from '../../../../../hooks/useVoluntariosDoMinisterioCrud';
8	import { zodResolver } from '@hookform/resolvers/zod';
9	import { useIndisponibilidadesVoluntariosCrud } from '../../../../../hooks/useIndisponibilidadesVoluntariosCrud';
10	import { useMemo, useState } from 'react';
11	import { Conjunction, Operator, ValueType } from '../../../../../domain/utils/query_utils';
12	import FancyError from '../../../../../components/error/FancyError';
13	import FancyLoading from '../../../../../components/FancyLoading';
14	import { Pallete } from '../../../../../constants/colors';
15	import FancyCalendar from '../../../../../components/calendar/FancyCalendar';
16	import FancyButton from '../../../../../components/buttons/FancyButton';
17	import { DefaultIconsNames } from '../../../../../constants/icons';
18	import FancyVerticalSpacer from '../../../../../components/FancyVerticalSpacer';
19	import FancyListEmpty from '../../../../../components/list/FancyListEmpty';
20	import AddPeriodoModal from '../../../../../components/pages/pessoal/indisponibilidade/AddPeriodModal';
21	import Toast from 'react-native-toast-message';
22	import DateUtils, { DateUtilsApi } from '../../../../../utils/date_utils';
23	import DateAvailabilityAdjustmentModal from '../../../../../components/pages/pessoal/indisponibilidade/DateAvailabilityAdjustmentModal';
24	import { useAuth } from '../../../../../contexts/AuthContext';
25	
26	type ModalState = {
27	  visible: boolean;
28	  date?: Date;
29	  status?: 'available' | 'unavailable';
30	  motivo?: string | null;
31	};
32	
33	const schema = z.object({
34	  voluntarioId: z.string('Campo obrigatório').min(1, 'Selecione um voluntário'),
35	  indisponibilidades: z.array(z.date()).min(1, 'Selecione ao menos uma indisponibilidade'),
36	});
37	
38	export default function MinisterioIndisponibilidadesIndex() {
39	  const { ministerioId } = useLocalSearchParams<{ ministerioId?: string }>();
40	  const { igrejaAtiva } = useAuth();
41	  const igrejaId = igrejaAtiva?.id;
42	
43	  const {
44	    voluntariosList,
45	    voluntariosDropDownList,
46	    isLoading: isLoadingVoluntarios,
47	  } = useVoluntariosDoMinisterioCrud(ministerioId);
48	
49	  const { control, watch } = useForm({
50	    resolver: zodResolver(schema),
51	    defaultValues: {
52	      voluntarioId: voluntariosList[0]?.id,
53	      indisponibilidades: [],
54	    },
55	  });
56	
57	  const voluntarioId = watch('voluntarioId');
58	
59	  const initialParams = useMemo(() => {
60	    if (!voluntarioId || !igrejaId) return undefined;
61	    return {
62	      igrejaId: igrejaId,
63	      where: {
64	        conditions: [
65	          {
66	            path: 'voluntario.id',
67	            operator: Operator.EQUALS,
68	            value: { type: ValueType.LITERAL, value: voluntarioId },
69	          },
70	        ],
71	        conjunction: Conjunction.AND,
72	      },
73	    };
74	  }, [voluntarioId, igrejaId]);
75	
76	  const {
77	    data: indisponibilidadesData,
78	    add: addIndisponibilidade,
79	    update: updateIndisponibilidade,
80	    removeWithIgreja,
81	    isLoading: isLoadingIndisponibilidades,
82	    isLoadingMutation: isLoadingIndisponibilidadesMutation,
83	    isError: isErrorIndisponibilidades,
84	    refetch: refetchIndisponibilidades,
85	    upsertMany,
86	  } = useIndisponibilidadesVoluntariosCrud({
87	    autoFetch: Boolean(voluntarioId && igrejaId),
88	    initialParams,
89	  });
90	
91	  const { startDate, endDate } = useMemo(() => {
92	    const now = new Date();
93	    return {
94	      startDate: new Date(now.getFullYear(), now.getMonth() - 1, now.getDate()),
95	      endDate: new Date(now.getFullYear(), now.getMonth() + 6, now.getDate()),
96	    };
97	  }, []);
98	
99	  const [modalState, setModalState] = useState<ModalState>({
100	    visible: false,
101	    status: 'available',
102	  });
103	  const [showPeriodoModal, setShowPeriodoModal] = useState(false);
104	
105	  const handleAddPeriodo = async (inicio: Date, fim: Date, motivo: string) => {
106	    setShowPeriodoModal(false);
107	    if (!voluntarioId || !igrejaId) return;
108	    try {
109	      const dates = DateUtils.generateDatesBetween(inicio, fim);
110	      const indisponibilidadesExistentes = dates.map((d) => ({
111	        data: DateUtilsApi.dateOnlyToApi(d),
112	        motivo: motivo?.trim() || undefined,
113	      }));
114	
115	      await upsertMany({
116	        voluntarioId,
117	        igrejaId,
118	        indisponibilidades: indisponibilidadesExistentes,
119	      });
120	      Toast.show({ type: 'success', text1: 'Período registrado com sucesso' });
121	    } catch {
122	      Toast.show({ type: 'error', text1: 'Erro ao registrar período' });
123	    }
124	  };
125	
126	  const closeModal = () => setModalState((s) => ({ ...s, visible: false }));
127	
128	  const handleConfirm = (mode: 'mark' | 'unmark', date: Date, motivo?: string) => {
129	    const registro = indisponibilidadesData.find((d) => new Date(d.data).getTime?.() === date.getTime?.());
130	
131	    closeModal();
132	
133	    (async () => {
134	      try {
135	        if (mode === 'mark') {
136	          if (registro?.id) {
137	            await updateIndisponibilidade({
138	              id: registro.id,
139	              data: {
140	                data: DateUtilsApi.dateOnlyToApi(date),
141	                motivo: motivo,
142	              },
143	            });
144	            Toast.show({
145	              type: 'success',
146	              text1: 'Motivo atualizado com sucesso!',
147	            });
148	          } else {
149	            if (!igrejaId) return;
150	            await addIndisponibilidade({
151	              data: DateUtilsApi.dateOnlyToApi(date),
152	              voluntarioId: voluntarioId,
153	              igrejaId,
154	              motivo,
155	            });
156	            Toast.show({
157	              type: 'success',
158	              text1: 'Data marcada como indisponível!',
159	            });
160	          }
161	        } else if (registro?.id && igrejaId) {
162	          await removeWithIgreja({ id: registro.id, igrejaId });
163	          Toast.show({
164	            type: 'info',
165	            text1: 'Data disponível novamente!!',
166	          });
167	        }
168	      } catch {
169	        Toast.show({ type: 'error', text1: 'Erro ao atualizar data!' });
170	      }
171	    })();
172	  };
173	
174	  if (isLoadingIndisponibilidades || isLoadingVoluntarios || isLoadingIndisponibilidadesMutation) {
175	    return <FancyLoading />;
176	  }
177	
178	  if (isErrorIndisponibilidades) {
179	    return <FancyError.Default onUpdate={refetchIndisponibilidades} />;
180	  }
181	
182	  return (
183	    <FancyPageView style={styles.container}>
184	      <View style={styles.voluntarioContainer}>
185	        <ControlledSearchSelect control={control} name='voluntarioId' label='Voluntário' searchPlaceholder='Buscar voluntário...' listItems={voluntariosDropDownList} />
186	      </View>
187	      {voluntarioId && !isLoadingIndisponibilidades ? (
188	        <View>
189	          <FancyVerticalSpacer height={15} />
190	          <FancyCalendar
191	            selectDateOnPress={false}
192	            onChangeSelectedDate={(date) => {
193	              const registro = indisponibilidadesData.find((d) => new Date(d.data).getTime() === date.getTime());
194	              setModalState({
195	                visible: true,
196	                date,
197	                status: registro ? 'unavailable' : 'available',
198	                motivo: registro?.motivo ?? null,
199	              });
200	            }}
201	            containerStyle={{ paddingHorizontal: 5 }}
202	            minimumDate={startDate}
203	            maximumDate={endDate}
204	            markedDates={indisponibilidadesData.map((d) => ({
205	              date: new Date(d.data),
206	              T: d.id,
207	              color: Pallete.error,
208	            }))}
209	            markedDatesType='SurroundCircle'
210	          />
211	          <FancyVerticalSpacer height={15} />
212	          <FancyButton
213	            label='Adicionar Período'
214	            icon={{ ...DefaultIconsNames.add, size: 20 }}
215	            onPress={() => setShowPeriodoModal(true)}
216	          />
217	        </View>
218	      ) : (
219	        <View
220	          style={{
221	            flex: 1,
222	            justifyContent: 'center',
223	            alignItems: 'center',
224	          }}
225	        >
226	          <FancyListEmpty label='Nenhum voluntário selecionado...' icon={{ library: 'MaterialCommunityIcons', name: 'calendar-remove-outline', size: 55 }} />
227	        </View>
228	      )}
229	
230	      {modalState.visible && modalState.date && modalState.status && (
231	        <DateAvailabilityAdjustmentModal
232	          data={{
233	            date: modalState.date,
234	            status: modalState.status,
235	            motivo: modalState.motivo ?? undefined,
236	          }}
237	          modalProps={{ onButton1Press: closeModal }}
238	          onConfirm={handleConfirm}
239	        />
240	      )}
241	
242	      {showPeriodoModal && (
243	        <AddPeriodoModal
244	          visible={showPeriodoModal}
245	          modalProps={{ onButton1Press: () => setShowPeriodoModal(false) }}
246	          onConfirm={handleAddPeriodo}
247	        />
248	      )}
249	    </FancyPageView>
250	  );
251	}
252	
253	const styles = StyleSheet.create({
254	  container: { paddingBottom: 10, flex: 1, paddingHorizontal: 20 },
255	  loadingOverlay: {
256	    ...StyleSheet.absoluteFillObject,
257	    backgroundColor: 'rgba(0,0,0,0.35)',
258	    justifyContent: 'center',
259	    alignItems: 'center',
260	    zIndex: 999,
261	  },
262	  legend: {
263	    flexDirection: 'row',
264	    paddingLeft: 20,
265	    alignItems: 'center',
266	    gap: 8,
267	  },
268	  legendCircle: {
269	    width: 14,
270	    height: 14,
271	    borderRadius: 7,
272	    backgroundColor: Pallete.error,
273	  },
274	  voluntarioContainer: { paddingTop: 5 },
275	});
