1	import FancyModalDialog, { FancyModalDialogProps } from '../../../../modal/FancyModalDialog';
2	import { useForm } from 'react-hook-form';
3	import { zodResolver } from '@hookform/resolvers/zod';
4	import ControlledSearchSelect from '../../../../forms/ControlledSearchSelect';
5	import ControlledBottomSheetSelect from '../../../../forms/ControlledBottomSheetSelect';
6	import { EnumUtils } from '../../../../../utils/enum_utils';
7	import ControlledNumberInput from '../../../../forms/ControlledNumberInput';
8	import { EscalaEventoTemplateFuncaoFormData, EscalaEventoTemplateFuncaoSchema } from '../../../../../domain/schemas/escalaSchema';
9	import { strfyObj } from '../../../../../utils/text_utils';
10	import { DropDownItemProps } from '../../../../fields/FancyDropDownItem';
11	import {
12	    EscalaTemplateExperienciaEnum,
13	    EscalaTemplateExperienciaLabel,
14	} from '../../../../../domain/enums/EscalaTemplate/escala-template-experiencia.enum';
15	
16	export interface EventoFormFuncaoModalProps {
17	  mode: 'add' | 'edit';
18	  data?: EscalaEventoTemplateFuncaoFormData;
19	  modalProps?: FancyModalDialogProps<EscalaEventoTemplateFuncaoFormData>;
20	  funcoesSelectionList: DropDownItemProps<string>[];
21	}
22	
23	export default function EventoFormFuncaoModal({ mode, data, modalProps, funcoesSelectionList }: EventoFormFuncaoModalProps) {
24	  const form = useForm<EscalaEventoTemplateFuncaoFormData>({
25	    resolver: zodResolver(EscalaEventoTemplateFuncaoSchema),
26	    defaultValues: data || { quantidade: 1 },
27	  });
28	
29	  return (
30	    <FancyModalDialog
31	      {...modalProps}
32	      title={mode === 'add' ? 'Nova Função' : 'Editar Função'}
33	      centerContainerStyle={{ gap: 12, paddingBottom: 5, paddingHorizontal: 5 }}
34	      onButton2Press={() => {
35	        form.handleSubmit(
36	          (data) => {
37	            modalProps?.onButton2Press?.(data);
38	          },
39	          (errors) => {
40	            console.log('Erro no formulário de adição de equipe', strfyObj(errors));
41	          },
42	        )();
43	      }}
44	    >
45	      <ControlledSearchSelect
46	        control={form.control}
47	        name='funcaoId'
48	        listItems={funcoesSelectionList}
49	        label='Função'
50	        disabled={mode === 'edit'}
51	        searchPlaceholder='Buscar função...'
52	      />
53	      <ControlledBottomSheetSelect
54	        control={form.control}
55	        name='experiencia'
56	        label='Experiência'
57	        listItems={EnumUtils.getDropDownItems(EscalaTemplateExperienciaEnum, EscalaTemplateExperienciaLabel)}
58	      />
59	      <ControlledNumberInput
60	        control={form.control}
61	        name='quantidade'
62	        title='Quantidade'
63	        min={1}
64	        max={99}
65	      />
66	    </FancyModalDialog>
67	  );
68	}
