1	import FancyModalDialog, { FancyModalDialogProps } from '../../../modal/FancyModalDialog';
2	import { useForm } from 'react-hook-form';
3	import ControlledPasswordInput from '../../../forms/ControlledPasswordInput';
4	import z from 'zod';
5	import { zodResolver } from '@hookform/resolvers/zod';
6	import { useAuth } from '../../../../contexts/AuthContext';
7	import Toast from 'react-native-toast-message';
8	import { AxiosError } from 'axios';
9	
10	const schema = z
11	  .object({
12	    senhaAtual: z.string().min(6, 'Senha atual inválida'),
13	    novaSenha: z.string().min(6, 'A nova senha deve ter pelo menos 6 caracteres'),
14	    confirmarSenha: z.string(),
15	  })
16	  .refine((data) => data.novaSenha === data.confirmarSenha, {
17	    message: 'As senhas não conferem',
18	    path: ['confirmarSenha'],
19	  });
20	
21	type FormData = z.infer<typeof schema>;
22	
23	export default function ChangePasswordModal(props: FancyModalDialogProps<any>) {
24	  const { changePassword } = useAuth();
25	  const {
26	    control,
27	    handleSubmit,
28	    formState: { isValid, isSubmitting },
29	  } = useForm<FormData>({
30	    resolver: zodResolver(schema),
31	    mode: 'onChange', // 👈 importante para validar em tempo real
32	  });
33	
34	  const handleConfirm = handleSubmit(async (data) => {
35	    try {
36	      const ok = await changePassword(data.senhaAtual, data.novaSenha);
37	      if (ok) {
38	        props.onButton2Press?.();
39	        Toast.show({ type: 'success', text1: 'Senha alterada com sucesso!' });
40	      } else {
41	        Toast.show({
42	          type: 'error',
43	          text1: 'Erro ao alterar a senha',
44	          text2: 'Verifique a senha atual e tente novamente.',
45	        });
46	      }
47	    } catch (e: AxiosError | any) {
48	      Toast.show({
49	        type: 'error',
50	        text1: 'Erro ao alterar a senha',
51	        text2: e?.response?.data?.message || 'Tente novamente.',
52	      });
53	    }
54	  });
55	
56	  return (
57	    <FancyModalDialog
58	      {...props}
59	      centerContainerStyle={{ gap: 15 }}
60	      title='Alterar Senha'
61	      onButton2Press={handleConfirm}
62	      button2={{
63	        disabled: !isValid || isSubmitting, // 👈 botão só habilita quando válido
64	        label: isSubmitting ? 'Alterando...' : 'Confirmar',
65	      }}
66	    >
67	      <ControlledPasswordInput control={control} name='senhaAtual' label='Senha Atual' />
68	      <ControlledPasswordInput control={control} name='novaSenha' label='Nova Senha' />
69	      <ControlledPasswordInput control={control} name='confirmarSenha' label='Confirmar Nova Senha' />
70	    </FancyModalDialog>
71	  );
72	}
